## 💻 CÓDIGO COMPLETO REESCRITO

```javascript
import fs from "fs/promises";
import path from "path";
import pm2 from "pm2";
import util from "util";
import { v4 as uuidv4 } from "uuid";

import { runDeepSeek, runGroq } from "../ai/index.js";
import { getDatabase } from "../database/sqlite.js";

const pm2Connect = util.promisify(pm2.connect.bind(pm2));
const pm2Start = util.promisify(pm2.start.bind(pm2));

const APPS_DIR = path.join(process.cwd(), "apps");

let currentPort = 4000;

// ─────────────────────────────────────────────
// 🧠 STEP 1: DeepSeek ANALYSIS
// ─────────────────────────────────────────────
async function analyzeIdea(idea) {
  return await runDeepSeek(`
You are the BRAIN of Geo AI.

Analyze this idea and convert it into a structured build instruction.

IDEA:
"${idea}"

Return ONLY JSON:
{
  "app_name": "",
  "description": "",
  "features": [],
  "pages": [],
  "api_endpoints": [],
  "complexity": "low | medium | high"
}
`);
}

// ─────────────────────────────────────────────
// 🧠 STEP 2: BUILD PROMPT FOR Z.AI
// ─────────────────────────────────────────────
function buildZaiPrompt(analysis, port) {
  return `
${ZAI_MASTER_PROMPT}

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
🧠 STRUCTURED INPUT (FROM DEEPSEEK)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

${analysis}

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
⚙️ SYSTEM CONTEXT
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

- Port: ${port}
- Environment: Node.js
- Execution: PM2 (Geo will handle)

Build the application now.
`;
}

// ─────────────────────────────────────────────
// 🧠 STEP 3: VALIDATE OUTPUT
// ─────────────────────────────────────────────
async function validateApp(jsonString) {
  return await runDeepSeek(`
You are a strict validator.

Check if this JSON is:
- Valid
- Complete
- Runnable

If VALID → return "OK"
If NOT → return FIXED JSON ONLY

${jsonString}
`);
}

// ─────────────────────────────────────────────
// 📦 INSTALL APP
// ─────────────────────────────────────────────
async function installApp(appData) {
  const appName = appData.name;
  const appPath = path.join(APPS_DIR, appName);

  await fs.mkdir(appPath, { recursive: true });

  for (const [file, content] of Object.entries(appData.frontend)) {
    await fs.writeFile(path.join(appPath, file), content);
  }

  for (const [file, content] of Object.entries(appData.backend)) {
    await fs.writeFile(path.join(appPath, file), content);
  }

  return appPath;
}

// ─────────────────────────────────────────────
// 🚀 PM2 LAUNCH
// ─────────────────────────────────────────────
async function launchApp(appName, appPath, port) {
  await pm2Connect();

  await pm2Start({
    script: path.join(appPath, "server.js"),
    name: appName,
    cwd: appPath,
    env: { PORT: port }
  });
}

// ─────────────────────────────────────────────
// 🔥 MAIN FUNCTION
// ─────────────────────────────────────────────
export async function createAppFromIdea(idea, tenantId = "default") {
  const port = ++currentPort;

  console.log("🧠 DeepSeek analyzing...");
  const analysis = await analyzeIdea(idea);

  console.log("🏗️ Building with Z.ai...");
  const zaiPrompt = buildZaiPrompt(analysis, port);

  let raw = await runDeepSeek(zaiPrompt); // Z.ai vía OpenRouter

  console.log("🧠 Validating...");
  let validation = await validateApp(raw);

  let appData;

  if (validation !== "OK") {
    appData = JSON.parse(validation);
  } else {
    appData = JSON.parse(raw);
  }

  // ⚡ FALLBACK GROQ SI FALLA
  if (!appData?.backend?.server.js) {
    console.log("⚡ Fallback to Groq...");
    const groqRaw = await runGroq(zaiPrompt);
    const match = groqRaw.match(/\{[\s\S]*\}/);
    appData = JSON.parse(match[0]);
  }

  console.log("📦 Installing...");
  const appPath = await installApp(appData);

  console.log("🚀 Launching...");
  await launchApp(appData.name, appPath, port);

  const db = await getDatabase();
  const id = uuidv4();

  await db.run(
    `INSERT INTO apps (id, name, description, port, tenant_id)
     VALUES (?, ?, ?, ?, ?)`,
    [id, appData.name, idea, port, tenantId]
  );

  return {
    id,
    name: appData.name,
    port
  };
}
```
